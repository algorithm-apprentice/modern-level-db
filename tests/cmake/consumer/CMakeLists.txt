cmake_minimum_required(VERSION 3.25)
project(modern_leveldb_consumer LANGUAGES CXX)

if(TEST_SCENARIO STREQUAL "module_shadow")
  list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/modules")
  set(MODERN_LEVELDB_BUILD_TESTS OFF CACHE BOOL "")
elseif(TEST_SCENARIO STREQUAL "default_tests")
  # Leave the library's options unset to check its subproject defaults.
elseif(TEST_SCENARIO STREQUAL "tests_disabled")
  set(MODERN_LEVELDB_BUILD_TESTS ON CACHE BOOL "")
  set(BUILD_TESTING OFF CACHE BOOL "")
elseif(TEST_SCENARIO STREQUAL "gtest_options")
  set(MODERN_LEVELDB_BUILD_TESTS ON CACHE BOOL "")
  set(BUILD_TESTING ON CACHE BOOL "")
  set(BUILD_GMOCK ON CACHE BOOL "")
  set(INSTALL_GTEST ON CACHE BOOL "")
  set(gtest_force_shared_crt OFF CACHE BOOL "")
else()
  message(FATAL_ERROR "Unknown consumer test scenario: ${TEST_SCENARIO}")
endif()

add_subdirectory("${MODERN_LEVELDB_SOURCE_DIR}" modern-leveldb)

if(NOT TARGET modern_leveldb::modern_leveldb)
  message(FATAL_ERROR "The consumer cannot link modern_leveldb::modern_leveldb")
endif()

if(TEST_SCENARIO STREQUAL "gtest_options")
  if(NOT BUILD_GMOCK OR NOT INSTALL_GTEST OR gtest_force_shared_crt)
    message(FATAL_ERROR "Modern LevelDB replaced the consumer's GoogleTest options")
  endif()
  if(NOT TARGET modern_leveldb_unit_tests OR NOT TARGET GTest::gmock)
    message(FATAL_ERROR "The consumer's requested test targets are missing")
  endif()
else()
  if(TARGET modern_leveldb_unit_tests OR TARGET GTest::gtest)
    message(FATAL_ERROR "A tests-disabled consumer acquired test dependencies")
  endif()
endif()
