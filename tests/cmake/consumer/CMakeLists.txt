cmake_minimum_required(VERSION 3.25)
project(modern_leveldb_consumer LANGUAGES CXX)

if(TEST_SCENARIO STREQUAL "module_shadow")
  list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/modules")
  set(MODERN_LEVELDB_BUILD_TESTS OFF CACHE BOOL "")
elseif(TEST_SCENARIO STREQUAL "default_tests")
  # Leave the library's options unset to check its subproject defaults.
elseif(TEST_SCENARIO STREQUAL "tests_disabled")
  set(MODERN_LEVELDB_BUILD_TESTS ON CACHE BOOL "")
  set(BUILD_TESTING OFF CACHE BOOL "")
elseif(TEST_SCENARIO STREQUAL "gtest_options")
  set(MODERN_LEVELDB_BUILD_TESTS ON CACHE BOOL "")
  set(BUILD_TESTING ON CACHE BOOL "")
  set(BUILD_GMOCK ON CACHE BOOL "")
  set(INSTALL_GTEST ON CACHE BOOL "")
  set(gtest_force_shared_crt OFF CACHE BOOL "")
elseif(TEST_SCENARIO STREQUAL "dependency_targets")
  set(MODERN_LEVELDB_BUILD_TESTS OFF CACHE BOOL "")
  add_library(parent_snappy INTERFACE)
  add_library(Snappy::snappy ALIAS parent_snappy)
  add_library(parent_zstd INTERFACE)
  add_library(zstd::libzstd_static ALIAS parent_zstd)
  add_library(parent_crc32c INTERFACE)
  add_library(Crc32c::crc32c ALIAS parent_crc32c)
elseif(TEST_SCENARIO STREQUAL "crc32c_parent_target")
  set(MODERN_LEVELDB_BUILD_TESTS OFF CACHE BOOL "")
  add_library(crc32c INTERFACE)
elseif(TEST_SCENARIO STREQUAL "crc32c_declaration")
  set(MODERN_LEVELDB_BUILD_TESTS OFF CACHE BOOL "")
  include(FetchContent)
  FetchContent_Declare(modern_leveldb_crc32c SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
elseif(TEST_SCENARIO STREQUAL "dependency_parent_values")
  set(MODERN_LEVELDB_BUILD_TESTS OFF CACHE BOOL "")
  set(BUILD_SHARED_LIBS ON CACHE BOOL "")
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "")
  set(ZSTD_BUILD_STATIC OFF CACHE BOOL "")
  set(ZSTD_BUILD_SHARED ON CACHE BOOL "")
  set(CRC32C_INSTALL ON CACHE BOOL "")
elseif(TEST_SCENARIO STREQUAL "dependency_shared_only")
  set(MODERN_LEVELDB_BUILD_TESTS OFF CACHE BOOL "")
  set(ZSTD_BUILD_STATIC OFF CACHE BOOL "")
elseif(TEST_SCENARIO STREQUAL "benchmark_reference_collision")
  set(MODERN_LEVELDB_BUILD_TESTS OFF CACHE BOOL "")
  set(MODERN_LEVELDB_BUILD_BENCHMARKS ON CACHE BOOL "")
  add_library(leveldb INTERFACE)
elseif(TEST_SCENARIO STREQUAL "profiling_framework_collision")
  set(MODERN_LEVELDB_BUILD_TESTS OFF CACHE BOOL "")
  set(MODERN_LEVELDB_BUILD_PERFORMANCE_TESTS ON CACHE BOOL "")
  add_library(benchmark INTERFACE)
else()
  message(FATAL_ERROR "Unknown consumer test scenario: ${TEST_SCENARIO}")
endif()

set(CXX_FLAGS_BEFORE "${CMAKE_CXX_FLAGS}")
set(REQUIRED_FLAGS_BEFORE "${CMAKE_REQUIRED_FLAGS}")
get_property(HAD_BUILD_SHARED_LIBS CACHE BUILD_SHARED_LIBS PROPERTY TYPE SET)
if(HAD_BUILD_SHARED_LIBS)
  set(BUILD_SHARED_LIBS_BEFORE "${BUILD_SHARED_LIBS}")
endif()
get_property(HAD_BUILD_TYPE CACHE CMAKE_BUILD_TYPE PROPERTY TYPE SET)
if(HAD_BUILD_TYPE)
  set(BUILD_TYPE_BEFORE "${CMAKE_BUILD_TYPE}")
endif()

add_subdirectory("${MODERN_LEVELDB_SOURCE_DIR}" modern-leveldb)

get_directory_property(
  library_cxx_flags DIRECTORY "${MODERN_LEVELDB_SOURCE_DIR}" DEFINITION CMAKE_CXX_FLAGS
)
if(NOT "${CMAKE_CXX_FLAGS}" STREQUAL "${CXX_FLAGS_BEFORE}" OR
   NOT "${library_cxx_flags}" STREQUAL "${CXX_FLAGS_BEFORE}" OR
   NOT "${CMAKE_REQUIRED_FLAGS}" STREQUAL "${REQUIRED_FLAGS_BEFORE}")
  message(FATAL_ERROR "Modern LevelDB changed shared compiler flags")
endif()
get_target_property(library_options modern_leveldb COMPILE_OPTIONS)
if("${library_options}" MATCHES "-march=|-msse4\\.2|/arch:")
  message(FATAL_ERROR "Modern LevelDB acquired dependency-specific ISA flags")
endif()

if(NOT TARGET modern_leveldb::modern_leveldb)
  message(FATAL_ERROR "The consumer cannot link modern_leveldb::modern_leveldb")
endif()

if(NOT TEST_SCENARIO STREQUAL "profiling_framework_collision")
  if(TARGET modern_leveldb_performance OR TARGET benchmark::benchmark)
    message(FATAL_ERROR "A performance-disabled consumer acquired Google Benchmark")
  endif()
endif()

get_property(HAS_BUILD_SHARED_LIBS CACHE BUILD_SHARED_LIBS PROPERTY TYPE SET)
if(NOT HAD_BUILD_SHARED_LIBS STREQUAL HAS_BUILD_SHARED_LIBS)
  message(FATAL_ERROR "Modern LevelDB changed whether BUILD_SHARED_LIBS exists")
endif()
if(HAD_BUILD_SHARED_LIBS AND NOT BUILD_SHARED_LIBS STREQUAL BUILD_SHARED_LIBS_BEFORE)
  message(FATAL_ERROR "Modern LevelDB replaced BUILD_SHARED_LIBS")
endif()
get_property(HAS_BUILD_TYPE CACHE CMAKE_BUILD_TYPE PROPERTY TYPE SET)
if(NOT HAD_BUILD_TYPE STREQUAL HAS_BUILD_TYPE)
  message(FATAL_ERROR "Modern LevelDB changed whether CMAKE_BUILD_TYPE exists")
endif()
if(HAD_BUILD_TYPE AND NOT CMAKE_BUILD_TYPE STREQUAL BUILD_TYPE_BEFORE)
  message(FATAL_ERROR "Modern LevelDB replaced CMAKE_BUILD_TYPE")
endif()

if(TEST_SCENARIO STREQUAL "gtest_options")
  if(NOT BUILD_GMOCK OR NOT INSTALL_GTEST OR gtest_force_shared_crt)
    message(FATAL_ERROR "Modern LevelDB replaced the consumer's GoogleTest options")
  endif()
  if(NOT TARGET modern_leveldb_unit_tests OR NOT TARGET GTest::gmock)
    message(FATAL_ERROR "The consumer's requested test targets are missing")
  endif()
else()
  if(TARGET modern_leveldb_unit_tests OR TARGET GTest::gtest)
    message(FATAL_ERROR "A tests-disabled consumer acquired test dependencies")
  endif()
endif()

if(TEST_SCENARIO STREQUAL "dependency_targets")
  if(TARGET snappy OR TARGET libzstd_static OR TARGET libzstd_shared OR TARGET crc32c)
    message(FATAL_ERROR "Modern LevelDB ignored parent codec targets")
  endif()
  get_target_property(library_dependencies modern_leveldb LINK_LIBRARIES)
  if(NOT "Crc32c::crc32c" IN_LIST library_dependencies)
    message(FATAL_ERROR "Modern LevelDB did not link the parent CRC32C target")
  endif()
elseif(TEST_SCENARIO STREQUAL "crc32c_parent_target")
  if(TARGET crc32c_arm64 OR TARGET crc32c_sse42)
    message(FATAL_ERROR "Modern LevelDB ignored the unnamespaced parent CRC32C target")
  endif()
  get_target_property(library_dependencies modern_leveldb LINK_LIBRARIES)
  if(NOT "crc32c" IN_LIST library_dependencies)
    message(FATAL_ERROR "Modern LevelDB did not link the parent CRC32C target")
  endif()
else()
  if(TARGET snappy_unittest OR TARGET snappy_benchmark OR TARGET crc32c_tests OR
     TARGET crc32c_bench OR TARGET glog OR TARGET zstd OR
     TARGET fullbench OR TARGET fuzzer OR TARGET zstreamtest)
    message(FATAL_ERROR "Modern LevelDB enabled dependency programs or tests")
  endif()
  get_target_property(crc32c_type crc32c TYPE)
  if(NOT crc32c_type STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "The fetched CRC32C dependency is not static")
  endif()
  if(TEST_SCENARIO STREQUAL "dependency_parent_values" AND NOT CRC32C_INSTALL)
    message(FATAL_ERROR "Modern LevelDB replaced the consumer's CRC32C option")
  endif()

  if(TEST_SCENARIO STREQUAL "dependency_shared_only")
    if(ZSTD_BUILD_STATIC OR NOT TARGET libzstd_shared OR TARGET libzstd_static)
      message(FATAL_ERROR "Modern LevelDB did not preserve the shared-only Zstd selection")
    endif()
    add_executable(codec_consumer main.cc)
    target_link_libraries(codec_consumer PRIVATE modern_leveldb::modern_leveldb)
  endif()
endif()
